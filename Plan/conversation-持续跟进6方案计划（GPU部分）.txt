    331 +  def _find_child_fd0(snippet, clip, harden_env):
    332 +      """在后台进程组里拉起读帧器（触发 L1 加固），回读 ffmpeg 子进程的 fd0。"""
    333 +      import json
    334 +      wrapper = _PREAMBLE + """
    335 +  os.setpgid(0, 0)                    # 后台进程组：L1 加固应当生效
    336 +  detach_background_stdin()
    337 +  """ + snippet
    338 +      p = subprocess.run(
    339 +          [sys.executable, "-c", wrapper, str(clip)],
    340 +          stdin=None, capture_output=True, text=True, timeout=180,
    341 +          cwd=str(ROOT), env=harden_env)
    342 +      if p.returncode != 0:
    343 +          raise RuntimeError("rc=%s stderr=%s" % (p.returncode, p.stderr[-600:]))
    344 +      lines = [x for x in p.stdout.strip().splitlines() if x.startswith("{")]
    345 +      if not lines:
    346 +          raise RuntimeError("无 JSON 输出：stdout=%r stderr=%r"
    347 +                             % (p.stdout[-400:], p.stderr[-400:]))
    348 +      return json.loads(lines[-1])

    364 +          results = {}

    366 +                                ("ESRGAN run_async", _ESRGAN_READER_SNIPPET)):
    367 +              try:
